```PRIMARY KEY: 唯一查找键, 可以为多列
CREATE TABLE Sailors(
	sid INTEGER,
	sname CHAR(20),
	rating_INTEGER,
	age_FLOAT,
	PRIMARY KEY (sid));
```

```FOREIGN KEY
CREATE TABLE Reserves(
	sid INTEGER,
	bid INTEGER,
	day_DATE,
	PRIMARY KEY (sid, bid, day),
	FOREIGN KEY_(sid)
	REFERENCES Sailors,
	FOREIGN KEY (bid)
	REFERENCES Boats);
```
